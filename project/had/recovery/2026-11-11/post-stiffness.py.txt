"""Small exact-frame comparison for the removable bridge support.
No foundation or anchor capacity is inferred. Units N/mm.
"""
import json, math
import numpy as np

E=200000.; G=E/2.6
nodes=np.array([[10140,3840,20],[10140,3840,2044],[10140,3840,2064],
                [10385,3970,2064],[9850,3700,20],[10150,3645,20]],float)
K=np.zeros((36,36))
def frame(a,b,w,t):
    d=nodes[b]-nodes[a]; L=np.linalg.norm(d); ex=d/L
    helper=np.array([0,0,1.]) if abs(ex[2])<.9 else np.array([1.,0,0])
    ey=np.cross(helper,ex); ey/=np.linalg.norm(ey); ez=np.cross(ex,ey)
    R=np.array([ex,ey,ez]); T=np.zeros((12,12))
    for j in range(4): T[j*3:j*3+3,j*3:j*3+3]=R
    A=w*w-(w-2*t)**2; I=(w**4-(w-2*t)**4)/12; J=(w-t)**3*t
    k=np.zeros((12,12))
    for inds,c in [([0,6],E*A/L),([3,9],G*J/L)]:
        k[np.ix_(inds,inds)]+=c*np.array([[1,-1],[-1,1]])
    base=np.array([[12,6*L,-12,6*L],[6*L,4*L*L,-6*L,2*L*L],
                   [-12,-6*L,12,-6*L],[6*L,2*L*L,-6*L,4*L*L]])*E*I/L**3
    k[np.ix_([1,5,7,11],[1,5,7,11])]+=base
    sig=np.diag([1,-1,1,-1]); k[np.ix_([2,4,8,10],[2,4,8,10])]+=sig@base@sig
    ix=np.r_[np.arange(6*a,6*a+6),np.arange(6*b,6*b+6)]
    K[np.ix_(ix,ix)]+=T.T@k@T
    return dict(a=int(a),b=int(b),L=float(L),section=[w,t],A=A,I=I,J=J)
def truss(a,b,w,t):
    d=nodes[b]-nodes[a]; L=np.linalg.norm(d); n=d/L; A=w*w-(w-2*t)**2
    q=E*A/L*np.outer(n,n); ix=np.r_[np.arange(6*a,6*a+3),np.arange(6*b,6*b+3)]
    K[np.ix_(ix,ix)]+=np.block([[q,-q],[-q,q]])
    return dict(a=int(a),b=int(b),L=float(L),section=[w,t],A=A,I=(w**4-(w-2*t)**4)/12,n=n)
members=[frame(0,1,80,4),frame(1,2,80,4),frame(2,3,40,4)]
braces=[truss(4,1,40,3),truss(5,1,40,3)]
# Fixed foot translations. The post base rotations are deliberately released
# in X/Y; only its physically anchored yaw restraint is retained. Brace ends
# are axial pins. This avoids silently crediting full floor moment fixity.
fixed=set([0,1,2,5]+list(range(24,36)))
free=np.array([i for i in range(36) if i not in fixed]); Kf=K[np.ix_(free,free)]
inv=np.linalg.inv(Kf)
def solve(f):
    u=np.zeros(36); u[free]=inv@f[free]; return u
# Obtain the full signed linear response to independently bounded force and
# moment vectors, not one favourable direction.
loads=[]; forces=[]
for dof,scale in [(18,250),(19,250),(20,500),(21,250000),(22,250000),(23,250000)]:
    f=np.zeros(36); f[dof]=scale; loads.append(solve(f)); forces.append(f)
U=np.array(loads).T
F=np.array(forces).T; reactions=K@U-F
force_balance=[]; moment_balance=[]
for case in range(6):
    forces_plus_reactions=(F[:,case]+reactions[:,case]).reshape(-1,6)
    force_balance.append(float(np.linalg.norm(forces_plus_reactions[:,:3].sum(axis=0))))
    moment_balance.append(float(np.linalg.norm((forces_plus_reactions[:,3:]+np.cross(nodes,forces_plus_reactions[:,:3])).sum(axis=0))))
reaction_bounds={}
for node in [0,4,5]:
    q=reactions[6*node:6*node+6]
    bound=np.linalg.norm(q[:,:2],axis=1)+np.abs(q[:,2])+np.linalg.norm(q[:,3:],axis=1)
    reaction_bounds[str(node)]=bound.tolist()

fork=U[18:21]
# Fxy has norm250, Fz has magnitude500, Mxyz has norm250Nm.
# Row norms give a conservative component displacement box; spectral norms
# provide a direct3D bound. No favourable cancellation is credited.
component=np.linalg.norm(fork[:,:2],axis=1)+np.abs(fork[:,2])+np.linalg.norm(fork[:,3:],axis=1)
norm_bound=np.linalg.norm(fork[:,:2],2)+np.linalg.norm(fork[:,2])+np.linalg.norm(fork[:,3:],2)
brace_results=[]
for b in braces:
    a,c=b['a'],b['b']; response=E*b['A']/b['L']*(b['n']@(U[6*c:6*c+3]-U[6*a:6*a+3]))
    demand=np.linalg.norm(response[:2])+abs(response[2])+np.linalg.norm(response[3:])
    euler=math.pi**2*E*b['I']/b['L']**2
    lam=math.sqrt(b['A']*355/euler); phi=.5*(1+.49*(lam-.2)+lam**2)
    chi=1/(phi+math.sqrt(phi**2-lam**2))
    brace_results.append({k:v for k,v in b.items() if k!='n'}|dict(
      forceBoundN=float(demand),EulerN=euler,comparisonAlpha=.49,slenderness=lam,
      bucklingReduction=chi,comparisonResistanceN=chi*b['A']*355/1.1,
      massKg=b['A']*b['L']*7.85e-6))
# Demand generated by a bounded0.1mm movement at either brace foot; foundation
# movement is not predicted. This is a required installed displacement budget.
foot_effect=[]
for node in [4,5]:
    for axis in range(3):
        prescribed=np.zeros(36); prescribed[6*node+axis]=.1
        u=prescribed.copy(); u[free]=-inv@(K[np.ix_(free,range(36))]@prescribed)
        foot_effect.append(float(np.linalg.norm(u[18:21])))
result=dict(status='BRACED_SUPPORT_COMPARISON_NOT_FLOOR_QUALIFICATION',nodes=nodes.tolist(),members=members,
  equilibriumMaxForceN=max(force_balance),equilibriumMaxMomentNmm=max(moment_balance),anchorStationComponentBounds=reaction_bounds,
  base='Translations and yaw fixed; base roll/pitch released. Actual floor/anchors must provide the stated displacement bounds.',
  forceBounds=dict(horizontalVectorN=250,verticalN=500,momentVectorNmm=250000),
  forkComponentDisplacementBoundMM=component.tolist(),forkVectorDisplacementBoundMM=float(norm_bound),
  signedLoadBasis=[dict(name=name,appliedN_Nmm=F[:,i].reshape(-1,6).tolist(),
    displacementsMM_rad=U[:,i].reshape(-1,6).tolist(),supportReactionsN_Nmm={str(n):reactions[6*n:6*n+6,i].tolist() for n in [0,4,5]})
    for i,name in enumerate(['Fx250','Fy250','Fz500','Mx250000','My250000','Mz250000'])],
  sixIndependent0_1mmBraceFootResponsesMM=foot_effect,
  allSixPrescribedAxisBoundsConservativeSumMM=sum(foot_effect),
  elasticPlusAllSixBoundsMM=float(norm_bound)+sum(foot_effect),
  worstOneFootResponseMM=max(foot_effect),braces=brace_results,
  fabricationPlacementBudgetMM=.5,positiveRemoteNominalGapMM=32.8,
  remainingIfOnlyOneFootMoves0_1mm=32.8-float(norm_bound)-max(foot_effect)-.5,
  qualification='Size anchors/base plates/pins from full signed reactions and prove installed translation/yaw compliance. These are comparison bounds, not predicted floor movements or measured assembly performance.')
with open(__file__.replace('.py','.json'),'w') as f: json.dump(result,f,indent=2)
print(json.dumps(result,indent=2))
